import React, { useState } from "react";
import { View, Text, TextInput, Button, FlatList } from "react-native";

export default function TradingJournal() {
  const [trades, setTrades] = useState([]);
  const [trade, setTrade] = useState({ asset: "", entry: "", exit: "", notes: "" });

  const addTrade = () => {
    if (trade.asset && trade.entry && trade.exit) {
      setTrades([...trades, { ...trade, id: Date.now().toString() }]);
      setTrade({ asset: "", entry: "", exit: "", notes: "" });
    }
  };

  return (
    <View style={{ padding: 20 }}>
      <Text style={{ fontSize: 24, fontWeight: "bold", marginBottom: 10 }}>Trading Journal</Text>
      <TextInput
        placeholder="Asset"
        value={trade.asset}
        onChangeText={(text) => setTrade({ ...trade, asset: text })}
        style={{ borderBottomWidth: 1, marginBottom: 10 }}
      />
      <TextInput
        placeholder="Entry Price"
        keyboardType="numeric"
        value={trade.entry}
        onChangeText={(text) => setTrade({ ...trade, entry: text })}
        style={{ borderBottomWidth: 1, marginBottom: 10 }}
      />
      <TextInput
        placeholder="Exit Price"
        keyboardType="numeric"
        value={trade.exit}
        onChangeText={(text) => setTrade({ ...trade, exit: text })}
        style={{ borderBottomWidth: 1, marginBottom: 10 }}
      />
      <TextInput
        placeholder="Notes"
        value={trade.notes}
        onChangeText={(text) => setTrade({ ...trade, notes: text })}
        style={{ borderBottomWidth: 1, marginBottom: 10 }}
      />
      <Button title="Add Trade" onPress={addTrade} />
      <FlatList
        data={trades}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <View style={{ padding: 10, borderBottomWidth: 1, marginTop: 10 }}>
            <Text>Asset: {item.asset}</Text>
            <Text>Entry: {item.entry}</Text>
            <Text>Exit: {item.exit}</Text>
            <Text>Notes: {item.notes}</Text>
          </View>
        )}
      />
    </View>
  );
}
